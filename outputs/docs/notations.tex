\begin{table}[h]
\centering
\caption{Table of notations. Calligraphic letters denote amortization scopes,
bold roman uppercase denote matrices, and bold italic lowercase denote vectors.}
\renewcommand{\arraystretch}{1.1}
\begin{tabularx}{\linewidth}{@{}p{2.4cm} l >{\raggedright\arraybackslash}X@{}}
\toprule
\textbf{Category} & \textbf{Symbol} &\textbf{Meaning} \\
\midrule
\multirow{4}{=}{\textit{Amortization scopes}}
 & $\mathcal{C}$ & Model class (root; e.g.\ sequential sampling models) \\
 & $\mathcal{F}$ & Model family (shared computational structure) \\
 & $\mathcal{M}$ & Model species / observation model (fully specified) \\
 & $\mathcal{D}$ & Design configuration over the intrinsic parameters \\
\midrule
\multirow{7}{=}{\textit{Generative objects}}
 & $\theta_{\mathcal{F}}$ & Intrinsic parameters of family $\mathcal{F}$ \\
 & $\rmB$ & Coefficient matrix, $\rmB \in \mathbb{R}^{R \times D_{\mathcal{F}}}$ (inference target) \\
 & $\rmM$ & Binary parameter mask, $\rmM \in \{0,1\}^{R \times D_{\mathcal{F}}}$; entries $M_{i,j}$ \\
 & $\rmX$ & Design matrix, $\rmX \in \mathbb{R}^{N \times R}$ \\
 & $\Theta$ & Realized parameter matrix, $\Theta = g(\rmX(\rmM \odot \rmB)) \in \mathbb{R}^{N \times D_{\mathcal{F}}}$ \\
 & $\rmY$ & Model observables, $\rmY \in \mathbb{R}^{N \times O_{\mathcal{F}}}$ \\
 & $\mathrm{RNG}$ & Source of stochastic variation in the observation model \\
\midrule
\multirow{11}{=}{\textit{Dimensions and indices}}
 & $N$ & Number of trials (observations) per dataset \\
 & $D_{\mathcal{F}}$ & Number of intrinsic parameters in family $\mathcal{F}$ \\
 & $O_{\mathcal{F}}$ & Number of observable dimensions in family $\mathcal{F}$ \\
 & $I$ & Number of regressors, indexed by $c$ \\
 & $R$ & Number of design-matrix columns, $R = I\,(C_{\mathrm{max}}-1)+1$ \\
 & $R_{\mathrm{max}}$ & Maximum column capacity (zero-padding target) \\
 & $C_{\mathrm{min}},\,C_{\mathrm{max}}$ & Bounds on the number of categories per regressor \\
 & $k_c$ & Realized number of categories of regressor $c$, $C_{\mathrm{min}}\le k_c\le C_{\mathrm{max}}$ \\
 & $i$ & Design-column index; row of $\rmB,\rmM$, with $i=0$ the intercept \\
 & $j$ & Intrinsic-parameter index; column of $\rmB,\rmM$ \\
 & $c$ & Regressor index; regressor $c$ spans a \emph{block} of design columns \\
\midrule
\multirow{3}{=}{\textit{Functions and parameters}}
 & $\mathcal{G}=\{g_j\}$ & Per-parameter link functions; $g$ denotes column-wise application \\
 & $\beta_{i,j}$ & Regression weight (entry of $\rmB$); $\beta_{0,j}$ is the intercept \\
 & $u_c$ & Regressor $c$ (continuous, or categorical with $k_c$ levels) \\
\midrule
\multirow{6}{=}{\textit{Design prior}}
 & $\bm{p}_c$ & Category marginal of categorical regressor $c$ \\
 & $\bm{P}$ & Joint category table over all categorical regressors \\
 & $\lambda$ & Association strength among categorical regressors, $\lambda\in[0,1]$ \\
 & $\eta^2$ & Correlation ratio of the categorical regressors on a continuous column \\
 & $\rmR$ & Residual correlation matrix of the continuous columns \\
 & $\zeta$ & LKJ shape parameter, $\rmR\sim\mathrm{LKJ}(\zeta)$ \\
\midrule
\multirow{4}{=}{\textit{Coefficient prior}}
 & $\sigma_j$ & Scale of the intercept prior for intrinsic parameter $j$ \\
 & $w_j$ & Design share: fraction of predictor variance that is design-driven \\
 & $\sigma^2_{\beta,j}$ & Slope variance budget, $\sigma^2_{\beta,j}=\sigma_j^2\,w_j/(1-w_j)$ \\
 & $\mu_k$ & Cell mean of category $k$; contrasts are $\beta_c=\mu_c-\mu_0$ \\
\midrule
\multirow{9}{=}{\textit{Flow matching and inference}}
 & $\mathcal{A}$ & Conditioning set, $\mathcal{A}=(\rmY,\rmX,\rmM,\mathcal{F})$ \\
 & $\hat{u}$ & Learned conditional velocity field, $\hat{u}(\rmZ_t;\mathcal{A},t)$ \\
 & $\rmZ_t$ & Flow state at time $t$, $\rmZ_t=(1-t)\rmB+t\rmZ_1$ \\
 & $\rmZ_0$ & Target endpoint, $\rmZ_0 \equiv \rmB$ \\
 & $\rmZ_1$ & Gaussian latent, $\rmZ_1\sim\mathcal{N}(\mathbf{0},\mathbf{I})$ \\
 & $t$ & Flow time, $t\in[0,1]$ \\
 & $\omega_t$ & Optional time-dependent loss weight \\
 & $\mathcal{L}_{\mathrm{FM}}$ & Conditional flow matching loss \\
 & $q(\rmB\mid\mathcal{A},\hat{u})$ & Approximate posterior induced by $\hat{u}$ \\
\midrule
\multirow{6}{=}{\textit{Architecture}}
 & $\rmS$ & Encoder summary representations, $\rmS=(\rvs_1,\dots,\rvs_L)$ \\
 & $\rvs_{\ell}$ & Summary vector, $\rvs_{\ell}\in\mathbb{R}^{S}$ \\
 & $L$ & Number of summary tokens (encoder seeds) \\
 & $S$ & Summary-representation dimension \\
 & $B_{i,j}$ & Decoder token for coefficient cell $(i,j)$ \\
 & $f$ & Integer index of family $\mathcal{F}$ (model-family embedding) \\
\bottomrule
\end{tabularx}
\label{tab:notation}
\end{table}
